        async function signup() {

            const email = document.getElementById('authEmail').value;

            const password = document.getElementById('authPassword').value;


            if (!email || !password) {

                showAuthError('Заполните email и пароль');

                return;

            }


            showAuthLoading(true);

            try {

                await auth.createUserWithEmailAndPassword(email, password);

                showAuthLoading(false);

            } catch (error) {

                showAuthLoading(false);

                showAuthError('Ошибка регистрации: ' + error.message);

            }

        }

        function updateAnalytics() {

            const period1 = data.payments.filter(p => {

                const date = new Date(p.date);

                return date.getDate() <= 15 && p.type === 'incoming';

            }).reduce((sum, p) => sum + p.amount, 0);


            const period2 = data.payments.filter(p => {

                const date = new Date(p.date);

                return date.getDate() > 15 && p.type === 'incoming';

            }).reduce((sum, p) => sum + p.amount, 0);


            const salary1 = data.payments.filter(p => {

                const date = new Date(p.date);

                return date.getDate() <= 15 && p.type === 'outgoing';

            }).reduce((sum, p) => sum + p.amount, 0);


            const salary2 = data.payments.filter(p => {

                const date = new Date(p.date);

                return date.getDate() > 15 && p.type === 'outgoing';

            }).reduce((sum, p) => sum + p.amount, 0);


            document.getElementById('period1Income').textContent = period1 + '₽';

            document.getElementById('period1Salary').textContent = salary1 + '₽';

            document.getElementById('period2Income').textContent = period2 + '₽';

            document.getElementById('period2Salary').textContent = salary2 + '₽';

        }

        function switchTab(tabName) {

            document.querySelectorAll('.tab-content').forEach(tab => {

                tab.classList.remove('active');

            });

            document.querySelectorAll('.nav-tab').forEach(btn => {

                btn.classList.remove('active');

            });

            document.getElementById(tabName).classList.add('active');

            event.target.classList.add('active');

        }
